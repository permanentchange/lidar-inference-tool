# Also usable as a standalone consumer after cmake --install.
cmake_minimum_required(VERSION 3.16)
project(bevfusion_cpp_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(bevfusion_cpp_api_example main.cpp)
if(TARGET inference)
  target_link_libraries(bevfusion_cpp_api_example PRIVATE inference)
  set_target_properties(bevfusion_cpp_api_example PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR};$ORIGIN")
  install(TARGETS bevfusion_cpp_api_example RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
else()
  find_package(lidar-inference-tool REQUIRED CONFIG)
  target_link_libraries(bevfusion_cpp_api_example PRIVATE lidar-inference-tool::inference)
endif()
set_target_properties(bevfusion_cpp_api_example PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
